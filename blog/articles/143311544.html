Problem A. 逆序对染色（思维+树状数组）

#include <bits/stdc++.h>

using namespace std;

#define int long long

struct BIT
{
	const int n;
	vector<int> tree;
	BIT(int n) : n(n), tree(n + 1) {};
	// 询问前x个数的和
	int Query(int x)
	{
		int res = 0;
		for (int i = x; i > 0; i -= (i & -i)) res += tree[i];
		return res;
	}
	// 第l个位置+z
	void Modify(int l, int z)
	{
		if (l <= 0) return;
		for (int i = l; i <= n; i += (i & -i)) tree[i] += z;
	}
	// 区间求和
	int rangeQuery(int l, int r)
	{
		return Query(min(n, r)) - Query(max(0ll, l - 1));
	}
};

void solve()
{
    int n; cin >> n;
    vector<int> a(n + 1), pos(n + 1);
    for (int i = 1; i <= n; i ++ )
    {
        cin >> a[i];
        pos[a[i]] = i;
    }
    vector<vector<int>> mem(n + 1);
    vector<bool> st(n + 1);
    for (int i = 1; i <= n; i ++ )
    {
        int t = pos[a[i] - 1] + 1;
        if (i >= t)
        {
            mem[t].push_back(i);
            st[a[i]] = true;
        }
    }
    BIT bit(n);
    int ans = 0;
    for (int i = 1; i <= n; i ++ )
    {
        for (auto t : mem[i]) bit.Modify(a[t], 1);
        if (st[a[i]]) bit.Modify(a[i], -1);
        if (a[i - 1] + 1 <= a[i] - 1) ans += bit.rangeQuery(a[i - 1] + 1, a[i] - 1);
    }
    cout << ans << '\n';
}

{

Problem G. 函数查询（二分+字典树）

  • 思路参考官方题解
#include <bits/stdc++.h>

using namespace std;

// #define int long long

typedef pair<int, int> PII;
const int INF = 1e9;

struct Trie
{
	// 输入s[i]后build() 根结点为1
	const static int ALPHABET = 2;
    const static int N = 9e6 + 5;
	struct Node
	{
        int maxx = -INF, minn = INF;
        int next[2]; // 子结点
    } tr[N];
	int n, cnt;
	vector<string> s; // 单词集合
	Trie(int n) : n(n), s(n + 1), cnt(1)
	{
		init();
	}

	void init()
	{
		for (int i = 0; i < ALPHABET; i ++ ) tr[0].next[i] = 1;
    }
    void insert(int idx, string &s)
	{
        int p = 1;
        for (auto c : s)
		{
		    tr[p].maxx = max(tr[p].maxx, idx);
		    tr[p].minn = min(tr[p].minn, idx);
            int x = c - '0';
            if (tr[p].next[x] == 0) tr[p].next[x] = ++ cnt;
            p = tr[p].next[x];
        }
		tr[p].maxx = max(tr[p].maxx, idx);
		tr[p].minn = min(tr[p].minn, idx);
    }
	void build()
	{
		for (int i = 1; i <= n; i ++ ) insert(i, s[i]);
	}
    PII find_bd(int a, int b, int op)
    {
        int mx = -INF, mn = INF;
        int p = 1, t;
        for (int i = 30; i >= 0; i -- )
        {
            int ab = ((a >> i) & 1);
            int bb = ((b >> i) & 1);
            t = ab ^ bb;
            if (op == 1 && bb == 0) mx = max(mx, tr[tr[p].next[t ^ 1]].maxx), mn = min(mn, tr[tr[p].next[t ^ 1]].minn);
            else if (op == 2 && bb == 1) mx = max(mx, tr[tr[p].next[t ^ 1]].maxx), mn = min(mn, tr[tr[p].next[t ^ 1]].minn);
            if (!tr[p].next[t])
            {
                p = tr[p].next[t];
                break;
            }
            if (i) p = tr[p].next[t];
        }
        if (p) mx = max(mx, tr[tr[p].next[t]].maxx), mn = min(mn, tr[tr[p].next[t]].minn);
        return make_pair(mn, mx);
    }
};

void solve()
{
    int n, q; cin >> n >> q;
    Trie trie(n);
    vector<int> x(n + 1);
    for (int i = 1; i <= n; i ++ )
    {
        cin >> x[i];
        for (int j = 30; j >= 0; j -- )
        {
            if ((x[i] >> j) & 1) trie.s[i] += '1';
            else trie.s[i] += '0';
        }
    }
    trie.build();

    while (q -- )
    {
        int a, b; cin >> a >> b;

        auto cal = [&](int x)
        {
            return (a ^ x) - b;
        };

        PII border_max = trie.find_bd(a, b, 1);
        PII border_min = trie.find_bd(a, b, 2);

        if (border_min.first <= border_max.second)
        {
            int l = border_min.first, r = border_max.second;
            while (l < r)
            {
                int mid = l + r >> 1;
                if (cal(x[mid]) >= 0) r = mid;
                else l = mid + 1;
            }
            cout << (r == 1 ? 1 : r - 1) << '\n';
        }
        else if (border_max.first <= border_min.second)
        {
            int l = border_max.first, r = border_min.second;
            while (l < r)
            {
                int mid = l + r >> 1;
                if (cal(x[mid]) <= 0) r = mid;
                else l = mid + 1;
            }
            cout << (r == 1 ? 1 : r - 1) << '\n';
        }
        else cout << -1 << '\n';
    }
}

signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(0), cout.tie(0);
    
    int t = 1;
    // cin >> t;
    while (t -- )
    {
        solve();
    }

Problem L. 毛肚下清汤？（签到）

#include <bits/stdc++.h>

using namespace std;

#define int long long

typedef pair<int, int> PII;

void solve()
{
    int n; cin >> n;
    priority_queue<PII, vector<PII>, greater<PII>> red, green;
    for (int i = 1; i <= n; i ++ )
    {
        int a, b, c, d; cin >> a >> b >> c >> d;
        if (c == 0 && d == 0) continue;
        else if (c == 1 && d == 0) red.push({a, i});
        else if (c == 0 && d == 1) green.push({b, i});
        else
        {
            if (a > b) green.push({b, i});
            else red.push({a, i});
        }
    }
    cout << red.size() << ' ';
    while (red.size())
    {
        auto t = red.top();
        red.pop();
        cout << t.second << ' ';
    }
    cout << '\n';
    cout << green.size() << ' ';
    while (green.size())
    {
        auto t = green.top();
        green.pop();
        cout << t.second << ' ';
    }
    cout << '\n';
}

signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(0), cout.tie(0);
